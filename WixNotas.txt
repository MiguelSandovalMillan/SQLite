--                                       _ATRIBUTOS DE DATOS_                                                  --

" NO NULL "  significa que los datos pueden repetirse

" PRIMARY KEY "  significa que el dato es unico (No se puede repetir)

" AUTOINCREMENT "  significa que el dato se va a ir sumando automaticamente (id = 1, id = 2, id = 3, etc)


--                                           _CONDICIONES_                                                     --

" WHERE " se utiliza para Condiciones  (WHERE coin = 'AGK') o (WHERE amount <= 20)

" AND " Nos ayuda para añadir mas condiciones a nuestra consulta (se tienen que cumplir todas las condiciones)  --> Prioritario

" OR " Nos ayuda para añadir mas condiciones a nuestra consulta (se tiene que cumplir al menos una de las condiciones)  --> No prioritario

" NOT " Nos ayuda para negar una condicion (WHERE NOT coin = 'AGK')

" TRUE " Se representa como 1 y significa que la condicion es (verdadera) --> Se utiliza con IS TRUE (WHERE coin IS TRUE)

" FALSE " Se representa como 0 y significa que la condicion es (falsa) --> Se utiliza con IS FALSE (WHERE coin IS FALSE)

" IS NULL " Se utiliza para saber si un campo esta vacio (WHERE coin IS NULL)

" IS NOT NULL " Se utiliza para saber si un campo no esta vacio (WHERE coin IS NOT NULL)

" IN " Se utiliza para saber si un campo esta dentro de un conjunto de datos (WHERE coin IN ('AGK', 'BTC', 'ETH'))

" BETWEEN " Se utiliza para saber si un campo esta entre un rango (WHERE amount BETWEEN 10 AND 20) --> Incluye los numeros 

" LIKE " Se utiliza para saber si un campo contiene un dato especifico (WHERE coin LIKE '%AGK%') 
        --> El simbolo "%" significa cualquier cantidad de caracteres 
        --> El simbolo "_" significa exactamenteun solo caracter   
            
            %a -> significa cualquier cadena que termine con "a".
            a% -> significa cualquier cadena que empiece con "a".
            %a% -> significa cualquier cadena que contenga "a".
            _a% -> significa que la letra "a" es el segundo carácter de la cadena.
            %a__ -> significa que la cadena contiene "a" en el tercer lugar desde el final.


" AS " Se utiliza para renombrar un campo (SELECT coin AS moneda FROM sales;)

--                                             _CONSULTAS_                                                     --

El simbolo  " * "  nos sirve para seleccionar todas las columnas de una tabla   (SELECT * FROM sales;)

La palabra " DISTINCT " para no repetir datos en la consulta y solo recibir los unicos  (SELECT DISTINCT coin FROM sales;)

La palabra " ORDER BY " para ordenar los datos de la consulta (SELECT * FROM sales ORDER BY coin;)

    -> Y se puede con " ASC " para ordenar de menor a mayor (SELECT * FROM sales ORDER BY coin ASC;)
    -> Y se puede con " DESC " para ordenar de mayor a menor (SELECT * FROM sales ORDER BY coin DESC;)

La palabra " LIMIT " para limitar la cantidad de datos que queremos recibir (SELECT * FROM sales LIMIT 10;)



Eliminar tablas " DROP TABLE "    (DROP TABLE sales;)


--                       _OPERADORES_                           --

" + "  Suma de numeros (SELECT amount + 10 FROM sales;)

" - "  Resta de numeros (SELECT amount - 10 FROM sales;)

" * "  Multiplicacion de numeros (SELECT amount * 10 FROM sales;)

" / "  Division de numeros (SELECT amount / 10 FROM sales;)

" % "  Modulo de numeros (SELECT amount % 10 FROM sales;)



            EJEMPLO
SELECT price + 10 as increased_price,
    price - 10 as reduced_price,
    price * 10 as multiple_price,
    price / 2 as half_price
FROM products;


--          _FUNCIONES_          --

" ROUND() "  Redondea un numero (SELECT ROUND(amount, 2) FROM sales;)
        --> ROUND(number, decimal_places) 
            number: El numero que quieres redondear
            decimal_places: El numero de posiciones decimales a los que se redondeara el numero
                - si es positivo redondea a ese número de posiciones decimales 
                - si es negativo redondea a la izquierda del punto decimal


                SELECT ROUND(3.7);                          SELECT ROUND(3.14159, 2);
                -- Returns 4.0                              -- Returns 3.14

                SELECT ROUND(3.3);                          SELECT ROUND(3.14159, 1);
                -- Returns 3.0                              -- Returns 3.1      

                SELECT ROUND(3.5);                          SELECT ROUND(3.14159, 0);
                -- Returns 4.0 (rounds up from .5)          -- Returns 3.0

                                                            SELECT ROUND(3.145, 2);
                                                            -- Returns 3.15 (rounds up because the next digit is 5)
