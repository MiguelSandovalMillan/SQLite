--                                                  ATRIBUTOS DE DATOS                                                  --

" NO NULL "  significa que los datos pueden repetirse

" PRIMARY KEY "  significa que el dato es unico (No se puede repetir)




--                                                     _CONDICIONES_                                                     --

" WHERE " se utiliza para Condiciones  (WHERE coin = 'AGK') o (WHERE amount <= 20)

" AND " Nos ayuda para añadir mas condiciones a nuestra consulta (se tienen que cumplir todas las condiciones)  --> Prioritario

" OR " Nos ayuda para añadir mas condiciones a nuestra consulta (se tiene que cumplir al menos una de las condiciones)  --> No prioritario

" NOT " Nos ayuda para negar una condicion (WHERE NOT coin = 'AGK')

" TRUE " Se representa como 1 y significa que la condicion es (verdadera) --> Se utiliza con IS TRUE (WHERE coin IS TRUE)

" FALSE " Se representa como 0 y significa que la condicion es (falsa) --> Se utiliza con IS FALSE (WHERE coin IS FALSE)

" IS NULL " Se utiliza para saber si un campo esta vacio (WHERE coin IS NULL)

" IS NOT NULL " Se utiliza para saber si un campo no esta vacio (WHERE coin IS NOT NULL)

" IN " Se utiliza para saber si un campo esta dentro de un conjunto de datos (WHERE coin IN ('AGK', 'BTC', 'ETH'))



--                                                       _CONSULTAS_                                                     --

El simbolo  " * "  nos sirve para seleccionar todas las columnas de una tabla   (SELECT * FROM sales;)

La palabra " DISTINCT " para no repetir datos en la consulta y solo recibir los unicos  (SELECT DISTINCT coin FROM sales;)

La palabra " ORDER BY " para ordenar los datos de la consulta (SELECT * FROM sales ORDER BY coin;)

    -> Y se puede con " ASC " para ordenar de menor a mayor (SELECT * FROM sales ORDER BY coin ASC;)
    -> Y se puede con " DESC " para ordenar de mayor a menor (SELECT * FROM sales ORDER BY coin DESC;)

La palabra " LIMIT " para limitar la cantidad de datos que queremos recibir (SELECT * FROM sales LIMIT 10;)



Eliminar tablas " DROP TABLE "    (DROP TABLE sales;)
